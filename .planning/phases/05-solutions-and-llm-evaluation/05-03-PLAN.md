  // Solutions (team-facing)
  async submitSolution(data: { content: string; solution_type: string }): Promise<SolutionResponse> {
    return this.request<SolutionResponse>("/api/v1/solutions/submit", {
      method: "POST",
      body: JSON.stringify(data),
    });
  }

  async getMySolutions(): Promise<SolutionResponse[]> {
    return this.request<SolutionResponse[]>("/api/v1/solutions/my-solutions");
  }

  async getCooldown(): Promise<CooldownResponse> {
    return this.request<CooldownResponse>("/api/v1/solutions/cooldown");
  }

  async getCaseline(): Promise<CaselineResponse> {
    return this.request<CaselineResponse>("/api/v1/solutions/caseline");
  }

  // Evaluations (admin-facing)
  async getEvaluations(status?: string): Promise<EvaluationListItem[]> {
    const query = status && status !== "ALL" ? `?status=${status}` : "";
    return this.request<EvaluationListItem[]>(`/api/v1/admin/evaluations${query}`);
  }

  async getEvaluation(evaluationId: string): Promise<EvaluationListItem> {
    return this.request<EvaluationListItem>(`/api/v1/admin/evaluations/${evaluationId}`);
  }

  async editEvaluation(evaluationId: string, data: Partial<{
    overall_score: number;
    scoring_metrics: string;
    news_headline: string;
    news_body: string;
    news_sentiment: number;
    price_timeline: string;
  }>): Promise<EvaluationActionResponse> {
    return this.request<EvaluationActionResponse>(`/api/v1/admin/evaluations/${evaluationId}`, {
      method: "PUT",
      body: JSON.stringify(data),
    });
  }

  async approveEvaluation(evaluationId: string): Promise<EvaluationActionResponse> {
    return this.request<EvaluationActionResponse>(`/api/v1/admin/evaluations/${evaluationId}/approve`, {
      method: "POST",

  }

    return this.request<EvaluationActionResponse>(`/api/v1/admin/evaluations/${evaluationId}/reject`, {

    caseline_document: string;

    });
  }

**A) Create client/src/stores/solutions-store.ts:**

Zustand store following existing patterns (see market-store.ts, trading-store.ts).

```typescript
import { create } from "zustand";
import type { SolutionResponse, CooldownResponse } from "@/lib/types";
import { api } from "@/lib/api";

interface SolutionsState {
  solutions: SolutionResponse[];
  cooldown: CooldownResponse | null;
  isSubmitting: boolean;
  isLoading: boolean;
  error: string | null;

  // Actions
  fetchSolutions: () => Promise<void>;
  fetchCooldown: () => Promise<void>;
  submitSolution: (content: string, solutionType: "INITIAL" | "MICROTWEAK") => Promise<void>;
  clearError: () => void;
}

export const useSolutionsStore = create<SolutionsState>((set, get) => ({
  solutions: [],
  cooldown: null,
  isSubmitting: false,
  isLoading: false,
  error: null,

  fetchSolutions: async () => {
    set({ isLoading: true });
    try {
      const solutions = await api.getMySolutions();
      set({ solutions, isLoading: false });
    } catch (err) {
      set({ error: err instanceof Error ? err.message : "Failed to load solutions", isLoading: false });
    }
  },

  fetchCooldown: async () => {
    try {
      const cooldown = await api.getCooldown();
      set({ cooldown });
    } catch {
      // Silent fail -- cooldown display is non-critical
    }
  },

  submitSolution: async (content, solutionType) => {
    set({ isSubmitting: true, error: null });
    try {
      await api.submitSolution({ content, solution_type: solutionType });
      // Refresh solutions list and cooldown after submit
      await get().fetchSolutions();
      await get().fetchCooldown();
      set({ isSubmitting: false });
    } catch (err) {
      set({ error: err instanceof Error ? err.message : "Submission failed", isSubmitting: false });
      throw err;  // Re-throw so UI can handle
    }
  },

  clearError: () => set({ error: null }),
}));
```
