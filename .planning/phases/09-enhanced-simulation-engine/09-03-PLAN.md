```
"""Celery task: compute fundamentals for one team at quarter boundary.

Mirrors evaluate_solution.py pattern. Fan-out dispatched by game_service
on Live Floor -> War Room transition (D-22).
"""
import asyncio, logging, uuid

logger = logging.getLogger(__name__)

FUNDAMENTAL_DELTA_SCHEMA = {
    "name": "fundamental_delta_output",
    "strict": True,
    "schema": {
        "type": "object",
        "properties": {
            "market_share_delta":   {"type": "number", "description": "Change in market share (percentage points). E.g. +1.2 means +1.2pp"},
            "margin_delta":         {"type": "number", "description": "Change in profitability margin (percentage points)"},
            "growth_delta":         {"type": "number", "description": "Change in brand-health proxy growth (percentage points)"},
            "ops_efficiency_delta": {"type": "number", "description": "Change in operational efficiency score (percentage points)"},
            "narrative":            {"type": "string", "description": "2-3 sentence rationale for the proposed deltas"},
        },
        "required": ["market_share_delta", "margin_delta", "growth_delta", "ops_efficiency_delta", "narrative"],
        "additionalProperties": False,
    },
}

from app.worker.celery_app import celery_app


@celery_app.task(
    name="fundamentals.compute_for_team",
    time_limit=120,
    soft_time_limit=100,
    max_retries=2,
    default_retry_delay=10,
    rate_limit="20/m",  # Pitfall 10 — throttle OpenRouter fan-out
)
def compute_fundamentals_for_team(round_number: int, team_id: str):
    asyncio.run(_compute(round_number, team_id))


async def _compute(round_number: int, team_id_str: str):
    from app.database import get_worker_session
    from app.models import FundamentalEvaluation, FundamentalEvalStatus
    from app.redis_pubsub import publish_broadcast_sync, publish_to_team_sync
    from app.services.fundamental_service import (
        build_fundamental_prompt, save_fundamental_evaluation,
    )
    from app.services.llm_service import call_openrouter

    tid = uuid.UUID(team_id_str)
    try:
        async with get_worker_session() as db:
            messages = await build_fundamental_prompt(db, tid, round_number)
            llm_output, metadata = await call_openrouter(
                messages, json_schema=FUNDAMENTAL_DELTA_SCHEMA
            )
            evaluation = await save_fundamental_evaluation(
                db, tid, round_number, llm_output, metadata
            )
            await db.commit()

        # Publish AFTER commit
        publish_broadcast_sync({
            "type": "fundamental_evaluation_pending",
            "payload": {
                "evaluation_id": str(evaluation.id),
                "team_id": str(evaluation.team_id),
                "round_number": round_number,
            },
        })
    except Exception as e:
        logger.error("compute_fundamentals failed for team %s: %s", team_id_str, e)
        try:
            async with get_worker_session() as fail_db:
                fail_db.add(FundamentalEvaluation(
                    team_id=tid,
                    round_number=round_number,
                    proposed_deltas="{}",
                    status=FundamentalEvalStatus.FAILED,
                    raw_response=str(e)[:5000],
                ))
                await fail_db.commit()
            publish_to_team_sync(tid, {
                "type": "fundamental_evaluation_failed",
                "payload": {"team_id": team_id_str, "round_number": round_number},
            })
        except Exception as inner:
            logger.error("Failed to record FAILED FundamentalEvaluation: %s", inner)
        raise  # Let Celery retry
```
