```python
async def _enqueue_shocks_from_news(
    db, payload: dict, source_ref_id
) -> int:
    """Enqueue PendingShock rows when a news payload has is_event=true.

    Returns the number of PendingShock rows written.
    Unknown tickers are logged and skipped (does not raise).
    """
    import uuid as _uuid
    from decimal import Decimal
    from sqlalchemy import select
    from app.models import PendingShock, ShockSource, Team
    from app.services.simulation_service import SEVERITY_MAG, DIRECTION_SIGN

    if not payload.get("is_event"):
        return 0
    severity = payload.get("severity")
    direction = payload.get("direction")
    tickers = payload.get("affected_tickers") or []
    if severity not in SEVERITY_MAG or direction not in DIRECTION_SIGN:
        logger.warning("Shock payload missing/invalid severity=%s direction=%s", severity, direction)
        return 0
    magnitude = SEVERITY_MAG[severity] * DIRECTION_SIGN[direction]

    count = 0
    for ticker in tickers:
        team = (await db.execute(select(Team).where(Team.ticker == ticker))).scalar_one_or_none()
        if team is None:
            logger.warning("Shock enqueue skipped — unknown ticker %s", ticker)
            continue
        db.add(PendingShock(
            team_id=team.id,
            magnitude_pct=magnitude,
            source=ShockSource.LLM_EVENT,
            source_ref_id=source_ref_id,
        ))
        count += 1
    return count
```
