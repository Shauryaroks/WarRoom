Per-tick formula (D-06):
  weighted_z = 0.35*z(market_share) + 0.30*z(margin) + 0.20*z(ops_efficiency) + 0.15*z(growth)

WEIGHTS: dict[str, Decimal] = {

    "margin":         Decimal("0.30"),
    "ops_efficiency": Decimal("0.20"),
    "growth":         Decimal("0.15"),

def get_current_quarter(elapsed_seconds: int) -> int:
    """Preserved from v1: 20-min windows = simulated quarters (D-04/SIM-02)."""

def z_score(value: Decimal, band_min: Decimal, band_max: Decimal) -> Decimal:
    """D-07: signed distance from band midpoint. 0 at midpoint, ±1 at edges, clamped ±2."""
    half_width = (band_max - band_min) / Decimal("2")
    if half_width == 0:
        return Decimal("0")
    midpoint = (band_min + band_max) / Decimal("2")
    z = (value - midpoint) / half_width
    return max(-Z_CLAMP, min(Z_CLAMP, z))


def compute_weighted_z(team: Team, bands: dict) -> Decimal:
    """Σ w_i * z_i. Null-safe — if any fundamental missing (Pitfall 4), returns 0."""
    for key in ("market_share", "margin", "growth", "ops_efficiency"):
        if getattr(team, f"fundamental_{key}") is None:
            return Decimal("0")
    return (
        WEIGHTS["market_share"]   * z_score(team.fundamental_market_share,    Decimal(str(bands["market_share"][0])),   Decimal(str(bands["market_share"][1])))
        + WEIGHTS["margin"]        * z_score(team.fundamental_margin,         Decimal(str(bands["margin"][0])),         Decimal(str(bands["margin"][1])))
        + WEIGHTS["ops_efficiency"] * z_score(team.fundamental_ops_efficiency, Decimal(str(bands["ops_efficiency"][0])), Decimal(str(bands["ops_efficiency"][1])))
        + WEIGHTS["growth"]        * z_score(team.fundamental_growth,         Decimal(str(bands["growth"][0])),         Decimal(str(bands["growth"][1])))
    )


def compute_drift_target(team: Team, bands: dict) -> Decimal:
    """D-06: clamp(0.15 * weighted_z, ±0.05)."""
    drift = DRIFT_COEF * compute_weighted_z(team, bands)
    return max(-DRIFT_CAP, min(DRIFT_CAP, drift))


def compute_ramp_fraction(team: Team, now: datetime) -> Decimal:
    """D-20: ramp_fraction = min(1.0, ticks_since_ramp_start / RAMP_TICKS). 0 if no ramp."""
    if team.ramp_started_at is None:
        return Decimal("0")
    ticks_since = int((now - team.ramp_started_at).total_seconds() // 60)
    if ticks_since <= 0:
        return Decimal("0")
    return min(Decimal("1.0"), Decimal(ticks_since) / Decimal(RAMP_TICKS))


def compute_price(
    team: Team,
    bands: dict,
    sentiment: Decimal,
    shock: Decimal,
    now: datetime,
) -> Decimal:
    """Per-tick price. Uses drift_floor_pct to avoid snap-back on ramp reset (D-21)."""
    ramp_frac = compute_ramp_fraction(team, now)
    target = team.drift_target_pct if team.drift_target_pct is not None else Decimal("0")
    natural = ramp_frac * target
    floor = team.drift_floor_pct if team.drift_floor_pct is not None else Decimal("0")
    # D-21: natural progression cannot drop below floor (the drift_now at prior reset)
    if target >= 0:
        drift_now = max(floor, natural)
    else:
        drift_now = min(floor, natural)

    noise_raw = Decimal(str(random.gauss(0, NOISE_SIGMA)))
    noise = max(-NOISE_BOUND, min(NOISE_BOUND, noise_raw))

    multiplier = Decimal("1") + drift_now + sentiment + shock + noise
    new_price = team.stock_price * multiplier
    return max(Decimal("0.01"), new_price.quantize(Decimal("0.01")))


async def calculate_all_prices(db: AsyncSession) -> tuple[list[dict], list[dict]]:
    """Per-tick price computation — batched, async, returns (price_updates, shock_events)."""
    now = datetime.now(timezone.utc)

    # 1) Batched SELECT: ACTIVE teams with archetype (skip teams without archetype, Pitfall 3)
    teams_result = await db.execute(
        select(Team)
        .options(selectinload(Team.archetype))
        .where(Team.status == TeamStatus.ACTIVE)
        .where(Team.archetype_id.is_not(None))
    )
    teams = teams_result.scalars().all()
    if not teams:
        return [], []

    # 2) Batched sentiment aggregate (D-13/D-14): last 60s signed INR volume by target_team_id
    cutoff = now - timedelta(seconds=60)
    signed_value = case(
        (Transaction.transaction_type == TransactionType.BUY,
         Transaction.shares * Transaction.price_per_share),
        (Transaction.transaction_type == TransactionType.SELL,
         -(Transaction.shares * Transaction.price_per_share)),
        else_=0,
    )
    sent_rows = await db.execute(
        select(
            Transaction.target_team_id,
            func.coalesce(func.sum(signed_value), 0).label("net_inr"),
        )
        .where(Transaction.executed_at >= cutoff)
        .where(Transaction.transaction_type.in_([TransactionType.BUY, TransactionType.SELL]))
        .group_by(Transaction.target_team_id)
    )
    sentiment_by_team: dict[uuid.UUID, Decimal] = {
        row.target_team_id: Decimal(str(row.net_inr)) for row in sent_rows
    }

    # 3) Pending shocks: FIFO one per team (D-19). Postgres DISTINCT ON pattern.
    shock_rows = await db.execute(
        select(PendingShock)
        .where(PendingShock.applied_at.is_(None))
        .order_by(PendingShock.team_id, PendingShock.enqueued_at)
        .distinct(PendingShock.team_id)
    )
    shock_by_team: dict[uuid.UUID, PendingShock] = {}
    for s in shock_rows.scalars().all():
        shock_by_team[s.team_id] = s

    price_updates: list[dict] = []
    shock_events: list[dict] = []
    applied_shock_ids: list[uuid.UUID] = []

    # 4) Per-team Python compute (pure-Python hot path, D-23)
    for team in teams:
        try:
            bands = json.loads(team.archetype.benchmark_bands)
        except (TypeError, ValueError, AttributeError):
            logger.warning("Team %s has invalid/missing benchmark_bands — skipping", team.id)
            continue

        # sentiment normalizer = stock_price × shares_outstanding × PUBLIC_FLOAT_RATIO
        net_inr = sentiment_by_team.get(team.id, Decimal("0"))
        public_float_value = team.stock_price * Decimal(team.shares_outstanding) * PUBLIC_FLOAT_RATIO
        if public_float_value > 0:
            sentiment_raw = net_inr / public_float_value
            sentiment = max(-SENTIMENT_CAP, min(SENTIMENT_CAP, sentiment_raw * SENTIMENT_COEF))
        else:
            sentiment = Decimal("0")

        # shock
        shock_obj = shock_by_team.get(team.id)
        if shock_obj is not None:
            shock = shock_obj.magnitude_pct / Decimal("100")
            applied_shock_ids.append(shock_obj.id)
            shock_events.append({
                "team_id": str(team.id),
                "ticker": team.ticker,
                "magnitude_pct": f"{shock_obj.magnitude_pct:.3f}",
                "source": shock_obj.source.value,
                "direction": "POSITIVE" if shock_obj.magnitude_pct >= 0 else "NEGATIVE",
            })
        else:
            shock = Decimal("0")

        old_price = team.stock_price
        new_price = compute_price(team, bands, sentiment, shock, now)
        team.stock_price = new_price

        # PriceHistory (D-24 — batch-insert via db.add; commit by caller)
        market_cap = new_price * Decimal(team.shares_outstanding)
        db.add(PriceHistory(
            team_id=team.id,
            price=new_price,
            market_cap=market_cap,
            calculated_at=now,
        ))

        change_pct = ((new_price - old_price) / old_price * Decimal("100")) if old_price > 0 else Decimal("0")
        price_updates.append({
            "team_id": str(team.id),
            "ticker": team.ticker,
            "price": str(new_price),
            "change_pct": f"{change_pct:+.2f}",
            "market_cap": str(market_cap),
            "timestamp": now.isoformat(),
        })

    # 5) Mark consumed shocks applied_at=now (audit trail retained, D-19)
    if applied_shock_ids:
        await db.execute(
            update(PendingShock)
            .where(PendingShock.id.in_(applied_shock_ids))
            .values(applied_at=now)
        )

    return price_updates, shock_events
