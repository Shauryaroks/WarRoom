@router.get("/fundamentals-queue", response_model=list[FundamentalEvalListItem])
async def list_fundamentals_queue(
    status_filter: Optional[str] = Query(None, alias="status"),
    _admin: dict = Depends(require_admin),
    db: AsyncSession = Depends(get_db),
):
    from sqlalchemy.orm import selectinload
    query = (
        select(FundamentalEvaluation)
        .options(selectinload(FundamentalEvaluation.team))
        .order_by(FundamentalEvaluation.created_at.desc())
    )
    if status_filter:
        try:
            s = FundamentalEvalStatus(status_filter)
            query = query.where(FundamentalEvaluation.status == s)
        except ValueError:
            raise HTTPException(
                status_code=400,
                detail=f"Invalid status: {status_filter}. Valid: {[s.value for s in FundamentalEvalStatus]}",
            )
    result = await db.execute(query)
    rows = result.scalars().all()
    return [
        FundamentalEvalListItem(
            id=str(e.id), team_id=str(e.team_id),
            team_name=e.team.name, team_ticker=e.team.ticker,
            round_number=e.round_number, status=e.status.value,
            proposed_deltas=e.proposed_deltas, admin_edits=e.admin_edits,
            narrative=e.narrative,
            approved_by_admin_id=str(e.approved_by_admin_id) if e.approved_by_admin_id else None,
            created_at=e.created_at.isoformat(),
            approved_at=e.approved_at.isoformat() if e.approved_at else None,
            rejected_at=e.rejected_at.isoformat() if e.rejected_at else None,
        ) for e in rows
    ]
```

@router.patch("/fundamentals-queue/{evaluation_id}", response_model=FundamentalActionResponse)
async def edit_fundamental_queue_item(
    evaluation_id: uuid.UUID,
    request: FundamentalEditRequest,
    _admin: dict = Depends(require_admin),
    db: AsyncSession = Depends(get_db),
):
    result = await db.execute(select(FundamentalEvaluation).where(FundamentalEvaluation.id == evaluation_id))
    evaluation = result.scalar_one_or_none()
    if not evaluation:
        raise HTTPException(status_code=404, detail="FundamentalEvaluation not found")
    if evaluation.status != FundamentalEvalStatus.PENDING_APPROVAL:
        raise HTTPException(status_code=409, detail=f"Cannot edit status={evaluation.status.value}")
    # Merge with existing admin_edits if any; otherwise fall back to proposed_deltas
    base = json.loads(evaluation.admin_edits) if evaluation.admin_edits else json.loads(evaluation.proposed_deltas)
    for key in ("market_share_delta", "margin_delta", "growth_delta", "ops_efficiency_delta"):
        v = getattr(request, key)
        if v is not None:
            base[key] = v
    evaluation.admin_edits = json.dumps(base)
    if request.narrative is not None:
        evaluation.narrative = request.narrative
    db.add(ActivityEvent(
        event_type="FUNDAMENTALS_EDITED",
        actor="admin",
        description=f"Edited fundamental eval {evaluation_id}",
        metadata_json=json.dumps({"evaluation_id": str(evaluation_id), "edits": base}),
    ))
    await db.commit()
    return FundamentalActionResponse(message="Edits saved", evaluation_id=str(evaluation_id), team_id=str(evaluation.team_id))
```

@router.post("/fundamentals-queue/{evaluation_id}/approve", response_model=FundamentalActionResponse)
async def approve_fundamental_endpoint(
    evaluation_id: uuid.UUID,
    _admin: dict = Depends(require_admin),
    db: AsyncSession = Depends(get_db),
):
    try:
        result = await apply_fundamental_deltas(db, evaluation_id, admin_id=uuid.UUID(_admin["admin_id"]))
    except ValueError as e:
        raise HTTPException(status_code=409, detail=str(e))
    db.add(ActivityEvent(
        event_type="FUNDAMENTALS_APPROVED",
        actor="admin",
        description=f"Fundamentals applied for {result['ticker']}",
        metadata_json=json.dumps(result),
    ))
    await db.commit()
    await publish_broadcast({
        "type": "fundamentals_updated",
        "payload": {
            "team_id": result["team_id"],
            "ticker": result["ticker"],
            "fundamentals": result["fundamentals"],
            "drift_target_pct": result["new_drift_target_pct"],
            "drift_floor_pct":  result["drift_floor_pct"],
        },
    })
    return FundamentalActionResponse(message="Fundamentals approved", evaluation_id=str(evaluation_id), team_id=result["team_id"])

        "ops_efficiency_delta": request.ops_efficiency_delta,
    }
    try:
        result = await override_fundamentals(db, team_id, overrides, admin_id=uuid.UUID(_admin["admin_id"]))
    except ValueError as e:
        raise HTTPException(status_code=400, detail=str(e))
    db.add(ActivityEvent(
        event_type="FUNDAMENTALS_OVERRIDDEN",
        actor="admin",
        description=f"Override applied to {result['ticker']}",
        metadata_json=json.dumps(result),
    ))
    await db.commit()
    await publish_broadcast({
        "type": "fundamentals_updated",
        "payload": {
            "team_id": result["team_id"],
            "ticker": result["ticker"],
            "fundamentals": result["fundamentals"],
            "drift_target_pct": result["new_drift_target_pct"],
            "drift_floor_pct":  result["drift_floor_pct"],
        },
    })
    return FundamentalActionResponse(message="Fundamentals overridden", evaluation_id="override", team_id=result["team_id"])
