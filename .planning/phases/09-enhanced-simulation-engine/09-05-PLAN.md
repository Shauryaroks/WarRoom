    id: str

    created_at: str            # ISO
    approved_at: Optional[str] = None

@router.get("/fundamentals-queue", response_model=list[FundamentalEvalListItem])
async def list_fundamentals_queue(
    status_filter: Optional[str] = Query(None, alias="status"),
    _admin: dict = Depends(require_admin),
    db: AsyncSession = Depends(get_db),
):
    from sqlalchemy.orm import selectinload
    query = (
        select(FundamentalEvaluation)
        .options(selectinload(FundamentalEvaluation.team))
        .order_by(FundamentalEvaluation.created_at.desc())
    )
    if status_filter:
        try:
            s = FundamentalEvalStatus(status_filter)
            query = query.where(FundamentalEvaluation.status == s)
        except ValueError:
            raise HTTPException(
                status_code=400,
                detail=f"Invalid status: {status_filter}. Valid: {[s.value for s in FundamentalEvalStatus]}",
            )
    result = await db.execute(query)
    rows = result.scalars().all()
    return [
        FundamentalEvalListItem(
            id=str(e.id), team_id=str(e.team_id),
            team_name=e.team.name, team_ticker=e.team.ticker,
            round_number=e.round_number, status=e.status.value,
            proposed_deltas=e.proposed_deltas, admin_edits=e.admin_edits,
            narrative=e.narrative,
            approved_by_admin_id=str(e.approved_by_admin_id) if e.approved_by_admin_id else None,
            created_at=e.created_at.isoformat(),
            approved_at=e.approved_at.isoformat() if e.approved_at else None,
            rejected_at=e.rejected_at.isoformat() if e.rejected_at else None,
        ) for e in rows
    ]
```

@router.patch("/fundamentals-queue/{evaluation_id}", response_model=FundamentalActionResponse)
async def edit_fundamental_queue_item(
    evaluation_id: uuid.UUID,
    request: FundamentalEditRequest,
    _admin: dict = Depends(require_admin),
    db: AsyncSession = Depends(get_db),
):
    result = await db.execute(select(FundamentalEvaluation).where(FundamentalEvaluation.id == evaluation_id))
    evaluation = result.scalar_one_or_none()
    if not evaluation:
        raise HTTPException(status_code=404, detail="FundamentalEvaluation not found")
    if evaluation.status != FundamentalEvalStatus.PENDING_APPROVAL:
        raise HTTPException(status_code=409, detail=f"Cannot edit status={evaluation.status.value}")
    base = json.loads(evaluation.admin_edits) if evaluation.admin_edits else json.loads(evaluation.proposed_deltas)
    for key in ("market_share_delta", "margin_delta", "growth_delta", "ops_efficiency_delta"):
        v = getattr(request, key)
        if v is not None:
            base[key] = v
    evaluation.admin_edits = json.dumps(base)
    if request.narrative is not None:
        evaluation.narrative = request.narrative
    db.add(ActivityEvent(
        event_type="FUNDAMENTALS_EDITED",
        actor="admin",
        description=f"Edited fundamental eval {evaluation_id}",
        metadata_json=json.dumps({"evaluation_id": str(evaluation_id), "edits": base}),
    ))
    await db.commit()
    return FundamentalActionResponse(message="Edits saved", evaluation_id=str(evaluation_id), team_id=str(evaluation.team_id))
```

    db: AsyncSession = Depends(get_db),
):
    try:
        result = await apply_fundamental_deltas(db, evaluation_id, admin_id=uuid.UUID(_admin["admin_id"]))
    except ValueError as e:

    db.add(ActivityEvent(
        event_type="FUNDAMENTALS_APPROVED",
        actor="admin",
        description=f"Fundamentals applied for {result['ticker']}",
        metadata_json=json.dumps(result),
    ))
    await db.commit()
    await publish_broadcast({
        "type": "fundamentals_updated",
        "payload": {
            "team_id": result["team_id"],
            "ticker": result["ticker"],
            "fundamentals": result["fundamentals"],
            "drift_target_pct": result["new_drift_target_pct"],
            "drift_floor_pct":  result["drift_floor_pct"],
        },
    })
    return FundamentalActionResponse(message="Fundamentals approved", evaluation_id=str(evaluation_id), team_id=result["team_id"])

class FundamentalRejectRequest(BaseModel):
    admin_notes: Optional[str] = None

@router.post("/fundamentals-queue/{evaluation_id}/reject", response_model=FundamentalActionResponse)
async def reject_fundamental_endpoint(
    evaluation_id: uuid.UUID,
    request: FundamentalRejectRequest,
    _admin: dict = Depends(require_admin),
    db: AsyncSession = Depends(get_db),
):
    try:
        result = await reject_fundamental_evaluation(
            db, evaluation_id, admin_id=uuid.UUID(_admin["admin_id"]),
            admin_notes=request.admin_notes,
        )
    except ValueError as e:
        raise HTTPException(status_code=409, detail=str(e))
    db.add(ActivityEvent(
        event_type="FUNDAMENTALS_REJECTED",
        actor="admin",
        description=f"Fundamental eval {evaluation_id} rejected",
        metadata_json=json.dumps({"evaluation_id": str(evaluation_id), "admin_notes": request.admin_notes}),
    ))
    await db.commit()
    await publish_broadcast({
        "type": "fundamentals_rejected",

def _build_team_fundamentals_response(team, bands, now):
    """Compute fundamentals response body for admin and team endpoints.

    SHARED HELPER — do not duplicate this logic in admin.py or teams.py. Callers
    must load Team with archetype eager-loaded and pass the parsed bands dict.

    Args:
        team: Team ORM instance (archetype pre-loaded by caller via selectinload)
        bands: dict parsed from team.archetype.benchmark_bands JSON
        now:   current datetime (UTC) for ramp_fraction

    Returns:
        TeamFundamentalsResponse (schemas/fundamentals.py)
    """
    from decimal import Decimal
    from app.schemas.fundamentals import TeamFundamentalsResponse, ZScoresResponse
    from app.services.simulation_service import z_score, compute_weighted_z, compute_ramp_fraction

    def _mid(key):
        return Decimal(str((bands[key][0] + bands[key][1]) / 2))

    z_ms = z_score(team.fundamental_market_share   if team.fundamental_market_share   is not None else _mid("market_share"),
                   Decimal(str(bands["market_share"][0])),   Decimal(str(bands["market_share"][1])))
    z_mg = z_score(team.fundamental_margin         if team.fundamental_margin         is not None else _mid("margin"),
                   Decimal(str(bands["margin"][0])),         Decimal(str(bands["margin"][1])))
    z_gr = z_score(team.fundamental_growth         if team.fundamental_growth         is not None else _mid("growth"),
                   Decimal(str(bands["growth"][0])),         Decimal(str(bands["growth"][1])))
    z_op = z_score(team.fundamental_ops_efficiency if team.fundamental_ops_efficiency is not None else _mid("ops_efficiency"),
                   Decimal(str(bands["ops_efficiency"][0])), Decimal(str(bands["ops_efficiency"][1])))

    return TeamFundamentalsResponse(
        team_id=str(team.id),
        ticker=team.ticker,
        fundamentals={
            "market_share":   str(team.fundamental_market_share)   if team.fundamental_market_share   is not None else None,
            "margin":         str(team.fundamental_margin)         if team.fundamental_margin         is not None else None,
            "growth":         str(team.fundamental_growth)         if team.fundamental_growth         is not None else None,
            "ops_efficiency": str(team.fundamental_ops_efficiency) if team.fundamental_ops_efficiency is not None else None,
        },
        bands=bands,
        z_scores=ZScoresResponse(
            z_market_share=str(z_ms), z_margin=str(z_mg),
            z_growth=str(z_gr), z_ops_efficiency=str(z_op),
        ),
        weighted_z=str(compute_weighted_z(team, bands)),
        drift_target_pct=str(team.drift_target_pct) if team.drift_target_pct is not None else None,
        drift_floor_pct=str(team.drift_floor_pct)   if team.drift_floor_pct   is not None else None,
        ramp_started_at=team.ramp_started_at.isoformat() if team.ramp_started_at else None,
        ramp_fraction=str(compute_ramp_fraction(team, now)),
    )
