**websocket.py** -- Rewrite the ConnectionManager. Remove ALL old burst_sync and dossier logic (those come back in Phase 2/3 with proper data). Keep it focused on connection management:

```python
import asyncio
import json
import logging
import uuid
from datetime import datetime, timezone
from typing import Dict, Set, Optional

from fastapi import WebSocket, WebSocketDisconnect

logger = logging.getLogger(__name__)


def _json_default(obj):
    if isinstance(obj, uuid.UUID):
        return str(obj)
    if isinstance(obj, datetime):
        return obj.isoformat()
    raise TypeError(f"Object of type {type(obj)} is not JSON serializable")


class ConnectionManager:
    """Manages WebSocket connections. Supports broadcast and per-team messaging."""

    def __init__(self):
        # All authenticated connections
        self._connections: Set[WebSocket] = set()
        # Team-specific connections: team_id -> set of WebSocket
        self._team_connections: Dict[uuid.UUID, Set[WebSocket]] = {}
        # Connection metadata: ws -> {"team_id": uuid, "role": str}
        self._metadata: Dict[WebSocket, dict] = {}

    @property
    def connection_count(self) -> int:
        return len(self._connections)

    def add(self, ws: WebSocket, team_id: Optional[uuid.UUID], role: str):
        self._connections.add(ws)
        self._metadata[ws] = {"team_id": team_id, "role": role}
        if team_id:
            if team_id not in self._team_connections:
                self._team_connections[team_id] = set()
            self._team_connections[team_id].add(ws)
        logger.info(f"WS connected: role={role}, team={team_id}, total={self.connection_count}")

    def remove(self, ws: WebSocket):
        meta = self._metadata.pop(ws, {})
        self._connections.discard(ws)
        team_id = meta.get("team_id")
        if team_id and team_id in self._team_connections:
            self._team_connections[team_id].discard(ws)
            if not self._team_connections[team_id]:
                del self._team_connections[team_id]
        logger.info(f"WS disconnected: role={meta.get('role')}, total={self.connection_count}")

    async def broadcast(self, message: dict):
        """Send message to ALL connected clients."""
        data = json.dumps(message, default=_json_default)
        await self._send_to_set(self._connections, data)

    async def send_to_team(self, team_id: uuid.UUID, message: dict):
        """Send message to all connections for a specific team."""
        connections = self._team_connections.get(team_id, set())
        if connections:
            data = json.dumps(message, default=_json_default)
            await self._send_to_set(connections, data)

    async def _send_to_set(self, connections: Set[WebSocket], data: str):
        """Send data to a set of connections, removing dead ones."""
        dead: list[WebSocket] = []
        tasks = []
        for ws in connections:
            tasks.append(self._safe_send(ws, data, dead))
        if tasks:
            await asyncio.gather(*tasks)
        for ws in dead:
            self.remove(ws)

    async def _safe_send(self, ws: WebSocket, data: str, dead: list):
        try:
            await ws.send_text(data)
        except Exception:
            dead.append(ws)


# Module-level singleton -- used by the API process only
manager = ConnectionManager()
```

**redis_pubsub.py** -- Redis pub/sub relay. This is the FIX for Anti-Pattern 1 (worker importing singleton):

```python
import asyncio
import json
import logging
import uuid

import redis.asyncio as aioredis

from app.database import settings
from app.websocket import manager

logger = logging.getLogger(__name__)

# Channel names
CHANNEL_BROADCAST = "ws:broadcast"
CHANNEL_TEAM_PREFIX = "ws:team:"


async def publish_broadcast(message: dict):
    """Publish a message to all WebSocket clients via Redis.
    Call this from Celery workers instead of importing the ConnectionManager."""
    r = aioredis.from_url(settings.REDIS_URL)
    try:
        await r.publish(CHANNEL_BROADCAST, json.dumps(message))
    finally:
        await r.aclose()


async def publish_to_team(team_id: uuid.UUID, message: dict):
    """Publish a message to a specific team's WebSocket clients via Redis."""
    r = aioredis.from_url(settings.REDIS_URL)
    try:
        await r.publish(f"{CHANNEL_TEAM_PREFIX}{team_id}", json.dumps(message))
    finally:
        await r.aclose()


def publish_broadcast_sync(message: dict):
    """Synchronous version for use in Celery tasks (sync context)."""
    import redis
    r = redis.from_url(settings.REDIS_URL)
    try:
        r.publish(CHANNEL_BROADCAST, json.dumps(message))
    finally:
        r.close()


def publish_to_team_sync(team_id: uuid.UUID, message: dict):
    """Synchronous version for use in Celery tasks (sync context)."""
    import redis
    r = redis.from_url(settings.REDIS_URL)
    try:
        r.publish(f"{CHANNEL_TEAM_PREFIX}{team_id}", json.dumps(message))
    finally:
        r.close()


async def start_redis_relay():
    """Subscribe to Redis channels and relay messages to WebSocket clients.
    Run this as a background task in the FastAPI lifespan."""
    r = aioredis.from_url(settings.REDIS_URL)
    pubsub = r.pubsub()

    await pubsub.subscribe(CHANNEL_BROADCAST)
    await pubsub.psubscribe(f"{CHANNEL_TEAM_PREFIX}*")

    logger.info("Redis pub/sub relay started")

    try:
        async for message in pubsub.listen():
            try:
                if message["type"] == "message":
                    channel = message["channel"]
                    if isinstance(channel, bytes):
                        channel = channel.decode()
                    data = json.loads(message["data"])

                    if channel == CHANNEL_BROADCAST:
                        await manager.broadcast(data)

                elif message["type"] == "pmessage":
                    channel = message["channel"]
                    if isinstance(channel, bytes):
                        channel = channel.decode()
                    data = json.loads(message["data"])

                    if channel.startswith(CHANNEL_TEAM_PREFIX):
                        team_id_str = channel[len(CHANNEL_TEAM_PREFIX):]
                        try:
                            team_id = uuid.UUID(team_id_str)
                            await manager.send_to_team(team_id, data)
                        except ValueError:
                            logger.warning(f"Invalid team_id in channel: {team_id_str}")

            except Exception as e:
                logger.error(f"Error processing Redis message: {e}")
    except asyncio.CancelledError:
        logger.info("Redis pub/sub relay shutting down")
    finally:
        await pubsub.unsubscribe()
        await pubsub.punsubscribe()
        await r.aclose()
```

   ```python
   @app.websocket("/ws")
   async def websocket_endpoint(ws: WebSocket):
       await ws.accept()

       # Step 1: Authenticate -- expect first message to be {"type": "auth", "token": "..."}
       try:
           auth_msg = await asyncio.wait_for(ws.receive_json(), timeout=10.0)
       except (asyncio.TimeoutError, Exception):
           await ws.close(code=4001, reason="Authentication timeout")
           return

       if auth_msg.get("type") != "auth" or "token" not in auth_msg:
           await ws.close(code=4002, reason="Expected auth message")
           return

       try:
           payload = decode_token(auth_msg["token"])
       except HTTPException:
           await ws.close(code=4003, reason="Invalid token")
           return

       role = payload.get("role", "")
       team_id = None
       if role in ("strategist", "trader"):
           team_id_str = payload.get("team_id")
           if team_id_str:
               team_id = uuid.UUID(team_id_str)

       # Step 2: Register connection
       manager.add(ws, team_id, role)

       # Step 3: Send welcome message
       await ws.send_json({
           "type": "connected",
           "payload": {
               "role": role,
               "team_id": str(team_id) if team_id else None,
               "connections": manager.connection_count,
           }
       })

       # Step 4: Listen for messages + ping/pong heartbeat
       try:
           while True:
               try:
                   data = await asyncio.wait_for(ws.receive_json(), timeout=30.0)
                   # Handle client messages (ping, sync_request, etc.)
                   if data.get("type") == "ping":
                       await ws.send_json({"type": "pong"})
               except asyncio.TimeoutError:
                   # Send server-initiated ping to keep connection alive through Caddy
                   try:
                       await ws.send_json({"type": "ping"})
                   except Exception:
                       break
       except WebSocketDisconnect:
           pass
       except Exception as e:
           logger.error(f"WebSocket error: {e}")
       finally:
           manager.remove(ws)
   ```

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-05-01 | Spoofing | WebSocket auth | mitigate | JWT required as first message within 10s timeout. Invalid/expired tokens close connection with code 4003. |
| T-05-02 | Denial of Service | WebSocket connections | mitigate | Connection count tracked in health endpoint. Server-initiated ping detects dead connections within 30s. Caddy zero-timeout prevents proxy-side kills. |
| T-05-03 | Information Disclosure | Redis pub/sub channels | accept | Redis bound to Docker internal network only. No external exposure. Messages contain game state, not credentials. |
| T-05-04 | Elevation of Privilege | WebSocket message injection | mitigate | Server only accepts "ping" and "sync_request" message types from clients. All state changes go through REST API. WebSocket is read-heavy (server push). |
