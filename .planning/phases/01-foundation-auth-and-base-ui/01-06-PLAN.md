**stores/auth-store.ts** -- Zustand store with persist middleware for session persistence (AUTH-06):

```typescript
import { create } from "zustand";
import { persist } from "zustand/middleware";
import type { UserRole } from "@/lib/types";

interface AuthState {
  token: string | null;
  teamId: string | null;
  teamName: string | null;
  ticker: string | null;
  role: UserRole | null;
  isAuthenticated: boolean;

  login: (data: {
    token: string;
    team_id: string;
    team_name: string;
    ticker: string;
    role: UserRole;
  }) => void;
  adminLogin: (token: string) => void;
  logout: () => void;
  setToken: (token: string) => void;
}

export const useAuthStore = create<AuthState>()(
  persist(
    (set) => ({
      token: null,
      teamId: null,
      teamName: null,
      ticker: null,
      role: null,
      isAuthenticated: false,

      login: (data) => {
        localStorage.setItem("token", data.token);
        set({
          token: data.token,
          teamId: data.team_id,
          teamName: data.team_name,
          ticker: data.ticker,
          role: data.role,
          isAuthenticated: true,
        });
      },

      adminLogin: (token) => {
        localStorage.setItem("token", token);
        set({
          token,
          teamId: null,
          teamName: null,
          ticker: null,
          role: "admin",
          isAuthenticated: true,
        });
      },

      logout: () => {
        localStorage.removeItem("token");
        set({
          token: null,
          teamId: null,
          teamName: null,
          ticker: null,
          role: null,
          isAuthenticated: false,
        });
      },

      setToken: (token) => {
        localStorage.setItem("token", token);
        set({ token });
      },
    }),
    {
      name: "complex-auth",
      partialize: (state) => ({
        token: state.token,
        teamId: state.teamId,
        teamName: state.teamName,
        ticker: state.ticker,
        role: state.role,
        isAuthenticated: state.isAuthenticated,
      }),
    }
  )
);
```

**hooks/use-websocket.ts** -- WebSocket hook with JWT auth handshake, reconnection with exponential backoff, and connection state tracking:

```typescript
"use client";

import { useEffect, useRef, useCallback, useState } from "react";
import { useAuthStore } from "@/stores/auth-store";

type MessageHandler = (data: Record<string, unknown>) => void;

interface UseWebSocketOptions {
  onMessage?: MessageHandler;
  enabled?: boolean;
}

interface UseWebSocketReturn {
  isConnected: boolean;
  sendMessage: (data: Record<string, unknown>) => void;
}

const WS_BASE = process.env.NEXT_PUBLIC_WS_URL || "ws://localhost:8080";

export function useWebSocket(options: UseWebSocketOptions = {}): UseWebSocketReturn {
  const { onMessage, enabled = true } = options;
  const token = useAuthStore((s) => s.token);
  const wsRef = useRef<WebSocket | null>(null);
  const [isConnected, setIsConnected] = useState(false);
  const reconnectAttempts = useRef(0);
  const maxReconnectAttempts = 10;
  const reconnectTimeoutRef = useRef<ReturnType<typeof setTimeout>>();
  const pingIntervalRef = useRef<ReturnType<typeof setInterval>>();
  const onMessageRef = useRef(onMessage);
  onMessageRef.current = onMessage;

  const connect = useCallback(() => {
    if (!token || !enabled) return;

    try {
      const ws = new WebSocket(`${WS_BASE}/ws`);
      wsRef.current = ws;

      ws.onopen = () => {
        // Send auth message as first message
        ws.send(JSON.stringify({ type: "auth", token }));

        // Start client-side ping every 25s (server expects within 30s)
        pingIntervalRef.current = setInterval(() => {
          if (ws.readyState === WebSocket.OPEN) {
            ws.send(JSON.stringify({ type: "ping" }));
          }
        }, 25000);
      };

      ws.onmessage = (event) => {
        try {
          const data = JSON.parse(event.data);

          if (data.type === "connected") {
            setIsConnected(true);
            reconnectAttempts.current = 0;
          } else if (data.type === "pong") {
            // Server responded to our ping -- connection alive
          } else if (data.type === "ping") {
            // Server-initiated ping -- respond with pong
            ws.send(JSON.stringify({ type: "pong" }));
          } else {
            onMessageRef.current?.(data);
          }
        } catch {
          // Ignore malformed messages
        }
      };

      ws.onclose = (event) => {
        setIsConnected(false);
        if (pingIntervalRef.current) clearInterval(pingIntervalRef.current);

        // Don't reconnect on auth failures (4001-4003)
        if (event.code >= 4001 && event.code <= 4003) return;

        // Exponential backoff reconnection
        if (reconnectAttempts.current < maxReconnectAttempts) {
          const delay = Math.min(1000 * Math.pow(2, reconnectAttempts.current), 30000);
          const jitter = Math.random() * 1000;
          reconnectAttempts.current++;
          reconnectTimeoutRef.current = setTimeout(connect, delay + jitter);
        }
      };

      ws.onerror = () => {
        // onclose will fire after onerror
      };
    } catch {
      // Connection failed -- onclose reconnect logic handles retry
    }
  }, [token, enabled]);

  useEffect(() => {
    connect();

    return () => {
      if (reconnectTimeoutRef.current) clearTimeout(reconnectTimeoutRef.current);
      if (pingIntervalRef.current) clearInterval(pingIntervalRef.current);
      if (wsRef.current) {
        wsRef.current.onclose = null; // prevent reconnect on intentional close
        wsRef.current.close();
      }
    };
  }, [connect]);

  const sendMessage = useCallback((data: Record<string, unknown>) => {
    if (wsRef.current?.readyState === WebSocket.OPEN) {
      wsRef.current.send(JSON.stringify(data));
    }
  }, []);

  return { isConnected, sendMessage };
}
```

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |

| T-06-01 | Spoofing | Session persistence | mitigate | Token stored via Zustand persist. JWT expiry enforced server-side. Logout clears all stored state. |
| T-06-02 | Information Disclosure | Auth state in localStorage | accept | Only token, team name, ticker, role stored. No passwords. Acceptable for competition app. |
| T-06-03 | Tampering | Client-side redirect bypass | accept | Protected route check is UX-only. All data access requires valid JWT on API calls. Client-side redirect prevents confusion, not unauthorized access. |
</threat_model>
