Add import for new types: `ArchetypeResponse, ArchetypeAssignResponse, ArchetypeSeedResponse`.

Add these methods to the ApiClient class:

```typescript
  // Archetype CRUD
  async getArchetypes(): Promise<ArchetypeResponse[]> {
    return this.request<ArchetypeResponse[]>("/api/v1/admin/archetypes");
  }

  async createArchetype(data: Omit<ArchetypeResponse, "id" | "is_preset" | "created_at">): Promise<ArchetypeResponse> {
    return this.request<ArchetypeResponse>("/api/v1/admin/archetypes", {
      method: "POST",
      body: JSON.stringify(data),
    });
  }

  async deleteArchetype(archetypeId: string): Promise<{ message: string }> {
    return this.request<{ message: string }>(`/api/v1/admin/archetypes/${archetypeId}`, {
      method: "DELETE",
    });
  }

  async assignArchetype(teamId: string, archetypeId: string): Promise<ArchetypeAssignResponse> {
    return this.request<ArchetypeAssignResponse>(`/api/v1/admin/teams/${teamId}/assign-archetype`, {
      method: "POST",
      body: JSON.stringify({ archetype_id: archetypeId }),
    });
  }

  async seedArchetypes(): Promise<ArchetypeSeedResponse> {
    return this.request<ArchetypeSeedResponse>("/api/v1/admin/archetypes/seed", {
      method: "POST",
    });
  }
```
