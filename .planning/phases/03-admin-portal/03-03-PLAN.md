Action handlers:
```typescript
async function handleApprove(teamId: string) {
  try {
    await api.approveTeam(teamId);
    adminStore.updateTeamStatus(teamId, "READY");
    toast.success("Team approved");
  } catch (err) {
    toast.error(err instanceof Error ? err.message : "Failed to approve team");
  }
}

async function handleReject(teamId: string) {
  try {
    await api.rejectTeam(teamId);
    adminStore.updateTeamStatus(teamId, "REJECTED");
    toast.success("Team rejected");
  } catch (err) {
    toast.error(err instanceof Error ? err.message : "Failed to reject team");
  }
}

async function handleRemove(teamId: string) {
  try {
    const result = await api.removeTeam(teamId);
    if (result.remove_type === "hard") {
      adminStore.removeTeam(teamId);
    } else {
      adminStore.updateTeamStatus(teamId, "BANKRUPT");
    }
    toast.success(result.message);
  } catch (err) {
    toast.error(err instanceof Error ? err.message : "Failed to remove team");
  }
}
```

WebSocket subscription (same pattern as dashboard):
- "team_registered": refresh team list
- "team_removed": update store

Both pages use `useAuthStore` for auth check and redirect, `useGameStore` for game status, `useAdminStore` for teams, `useWebSocket` for real-time updates.
