**2. Create client/src/stores/admin-store.ts:**
Zustand store WITHOUT persist middleware (ephemeral state, reloads on page visit):
```typescript
import { create } from "zustand";
import type { AdminTeam, NewsItem, ActivityEvent } from "@/lib/types";

interface AdminState {
  teams: AdminTeam[];
  news: NewsItem[];
  activityLog: ActivityEvent[];

  setTeams: (teams: AdminTeam[]) => void;
  updateTeamStatus: (teamId: string, status: string) => void;
  removeTeam: (teamId: string) => void;
  setNews: (news: NewsItem[]) => void;
  addNews: (item: NewsItem) => void;
  setActivityLog: (events: ActivityEvent[]) => void;
  addActivity: (event: ActivityEvent) => void;
}

export const useAdminStore = create<AdminState>()((set) => ({
  teams: [],
  news: [],
  activityLog: [],

  setTeams: (teams) => set({ teams }),
  updateTeamStatus: (teamId, status) =>
    set((s) => ({
      teams: s.teams.map((t) => (t.id === teamId ? { ...t, status } : t)),
    })),
  removeTeam: (teamId) =>
    set((s) => ({ teams: s.teams.filter((t) => t.id !== teamId) })),
  setNews: (news) => set({ news }),
  addNews: (item) => set((s) => ({ news: [item, ...s.news] })),
  setActivityLog: (events) => set({ activityLog: events }),
  addActivity: (event) =>
    set((s) => ({
      activityLog: [event, ...s.activityLog].slice(0, 200),
    })),
}));
```

// Team management
async getTeams(status?: string): Promise<AdminTeam[]> {
  const query = status ? `?status=${status}` : "";
  return this.request<AdminTeam[]>(`/api/v1/admin/teams${query}`);
}

async approveTeam(teamId: string): Promise<{ message: string; team_id: string; new_status: string }> {
  return this.request(`/api/v1/admin/teams/${teamId}/approve`, { method: "POST" });
}

async rejectTeam(teamId: string): Promise<{ message: string; team_id: string; new_status: string }> {
  return this.request(`/api/v1/admin/teams/${teamId}/reject`, { method: "POST" });
}

async removeTeam(teamId: string): Promise<TeamRemoveResponse> {
  return this.request<TeamRemoveResponse>(`/api/v1/admin/teams/${teamId}/remove`, { method: "POST" });
}

// News
async injectNews(data: { headline: string; body: string; news_type: string; affected_tickers: string[] }): Promise<NewsInjectResponse> {
  return this.request<NewsInjectResponse>("/api/v1/admin/news/inject", { method: "POST", body: JSON.stringify(data) });
}

async getNewsHistory(): Promise<NewsItem[]> {
  return this.request<NewsItem[]>("/api/v1/admin/news/history");
}

// Broadcast
async broadcast(message: string): Promise<BroadcastResponse> {
  return this.request<BroadcastResponse>("/api/v1/admin/broadcast", { method: "POST", body: JSON.stringify({ message }) });
}

// Activity
async getActivityLog(): Promise<ActivityEvent[]> {
  return this.request<ActivityEvent[]>("/api/v1/admin/activity");
}

// Round management
async advanceRound(): Promise<RoundAdvanceResponse> {
  return this.request<RoundAdvanceResponse>("/api/v1/admin/game/advance-round", { method: "POST" });
}
