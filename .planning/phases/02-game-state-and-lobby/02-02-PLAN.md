    ```typescript
    import { create } from "zustand";
    import type { GameStatus, GameStatePayload } from "@/lib/types";

    interface GameState {
      status: GameStatus;
      elapsedSeconds: number;
      currentRound: number;

      setGameState: (payload: GameStatePayload) => void;
      reset: () => void;
    }

    export const useGameStore = create<GameState>()((set) => ({
      status: "SETUP",
      elapsedSeconds: 0,
      currentRound: 0,

      setGameState: (payload) =>
        set({
          status: payload.status,
          elapsedSeconds: payload.elapsed_seconds,
          currentRound: payload.current_round,
        }),
      reset: () =>
        set({
          status: "SETUP",
          elapsedSeconds: 0,
          currentRound: 0,
        }),
    }));
    ```

    Do NOT use persist middleware for game store -- game state is transient and always received from server.

| T-02-08 | Tampering | Client timer manipulation | mitigate | D-07: Timer value comes from server via WebSocket. No client-side computation. Even if client modifies store, server state is unaffected. |
